def separate_by_class(dataset):
    is_class_0 = dataset["y"] == 0
    is_class_1 = ~is_class_0

    d1 = dataset[is_class_0].reset_index(drop=True)
    d2 = dataset[is_class_1].reset_index(drop=True)
    return d1, d2

def scatter_plot(X, plotline=False, offset=0.3, nombre = None):
    clase1,clase2 = separate_by_class(X)
    
    plt.scatter(clase1["x1"], clase1["x2"], label = "0")
    plt.scatter(clase2["x1"], clase2["x2"], label = "1")

    if plotline:
        x_min = min(np.min(clase1["x1"]), np.min(clase2["x2"]))
        y_min = min(np.min(clase1["x2"]), np.min(clase2["x2"]))

        rel_height = lambda x, y: (y-y_min)/(x-x_min)
        
        c1_min = np.argmin([rel_height(x,y) for x,y in zip(clase1["x1"], clase1["x2"])])
        c2_max = np.argmax([rel_height(x,y) for x,y in zip(clase2["x1"], clase2["x2"])])

        p1 = [clase1["x1"][c1_min], clase1["x2"][c1_min]-offset]
        p2 = [clase2["x1"][c2_max], clase2["x2"][c2_max]+offset]
        
        plt.scatter(p1[0], p1[1], marker="x")
        plt.scatter(p2[0], p2[1], marker="x")

        plt.axline(p1, p2)
        
    plt.legend()    
    plt.savefig(f"Scatter_{nombre}.png", bbox_inches='tight')
    plt.plot()
    
def plot_boundaries_and_support_vectors(X, estimator, draw_support_vectors=True, size=50, nombre = None):
    ax = plt.gca()
    DecisionBoundaryDisplay.from_estimator(
        estimator,
        X[["x1","x2"]].values,
        plot_method='contourf',
        colors=["blue","orange"],
        levels=[-1, 0, 1],
        alpha=0.4,
        linestyles=["--", "-", "--"],
        ax=ax,
    )

    clase1,clase2 = separate_by_class(X)
    plt.scatter(clase1["x1"], clase1["x2"], label = "0", color="blue", s=size)
    plt.scatter(clase2["x1"], clase2["x2"], label = "1", color="orange", s=size)

    if draw_support_vectors:
        support_vectors = estimator.support_vectors_
        plt.scatter(
            support_vectors[:,0],
            support_vectors[:,1],         
            s=int(size*1.10),
            linewidth=1,
            facecolors="none",
            edgecolors="k",
        )

    plt.title("Boundaries")
    
    plt.xlabel('x')
    plt.ylabel('y')

    plt.legend()    

    plt.savefig(f"SVM_{nombre}.png", bbox_inches='tight')
    plt.plot()

def generic_svm_cv(
    dataset,
    param_dicts,
    x_columns = ["x1","x2"],
    y_column = "y",
    in_split = 4,
    out_split = 3,
    metric = "f1",
):
    X = dataset[x_columns].values
    Y = dataset[y_column].values

    fuera_cv = StratifiedKFold(n_splits=out_split, shuffle=True, random_state=42)
    dentro_cv = StratifiedKFold(n_splits=in_split, shuffle=True, random_state=42)

    results_fuera = {
        "best_index": []
    }

    for i, (ixs_train_fuera, ixs_test_fuera) in enumerate(fuera_cv.split(X, Y)):
        # Datos de entrenamiento para el bucle exterior
        X_train_dentro = X[ixs_train_fuera]
        Y_train_dentro = Y[ixs_train_fuera] 

        multiple_fits = GridSearchCV(
            SVC(),
            param_dicts,
            cv = dentro_cv,
            scoring = metric,
        )

        multiple_fits.fit(X_train_dentro, Y_train_dentro)

        results_fuera["best_index"].append(multiple_fits.best_index_)
        results_fuera['params'] = multiple_fits.cv_results_["params"]

    results = {}
    
    best_index = st.mode(results_fuera["best_index"])
    best_model = SVC(**results_fuera["params"][best_index])

    cross_validation_fuera = cross_validate(
        best_model,
        X, Y,
        cv = fuera_cv,
        return_estimator = True,
        scoring = metric,
    )

    results = {
        'best': cross_validation_fuera["estimator"][np.argmax(cross_validation_fuera["test_score"])],
        'params': results_fuera["params"][best_index],
        'scores': cross_validation_fuera["test_score"],
        "mean_score": st.mean(cross_validation_fuera["test_score"])
    }

    return results

def Generic_simple_cv(
    dataset,
    param_dicts,
    x_columns = ["x1","x2"],
    y_column = "y",
    out_split = 5,
    metric = "f1",
):
    X = dataset[x_columns].values
    Y = dataset[y_column].values

    fuera_cv = StratifiedKFold(n_splits=out_split, shuffle=True, random_state=42)

    cross_validation_fuera = GridSearchCV(
        SVC(),
        param_dicts,
        cv = fuera_cv,
        scoring = metric,
        verbose=2,
        n_jobs=4,
    )

    cross_validation_fuera.fit(X, Y)

    best_index = cross_validation_fuera.best_index_

    results = {
        'best': cross_validation_fuera.best_estimator_,
        'params': cross_validation_fuera.cv_results_["params"][best_index],
        'scores': cross_validation_fuera.cv_results_["mean_test_score"],
        'best_score': cross_validation_fuera.best_score_,
    }

    return results

def normalized_df(df, columns):
    mean = df[columns].mean()
    std = df[columns].std()
    
    return ((df[columns]-mean) / std)
